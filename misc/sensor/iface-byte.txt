sensor ifaces-hw
path interfaces-hw/interface/counter
prefix freertr-ifaces
key name interfaces-hw/interface
command sho inter hwsumm | linenumbers
prepend iface_hw_byte_
name 1 ifc=
oid-begin 2
oid-name 1.3.6.1.2.1.2.2.1.2 tag=octetstring string=
replace \. _
skip 1
column 0 oid 1.3.6.1.2.1.2.2.1.1 tag=2 int32=
column 2 name st
column 2 replace admin -1
column 2 replace down 0
column 2 replace up 1
column 3 name tx
column 3 oid 1.3.6.1.2.1.31.1.1.1.10 tag=6 class=1 int64=
column 4 name rx
column 4 oid 1.3.6.1.2.1.31.1.1.1.6 tag=6 class=1 int64=
column 5 name dr
column 5 oid 1.3.6.1.2.1.2.2.1.13 tag=1 class=1 int32=
.
exit
