sensor ifaces-status-hw
path interfaces-status-hw/interface/counter
prefix freertr-ifaces
command sho inter hwstatus | linenumbers
name 1 ifc=
oid-begin 2
oid-name 1.3.6.1.2.1.2.2.1.2 tag=octetstring string=
key name interfaces-status-hw/interface
replace \. _
skip 1
column 0 oid 1.3.6.1.2.1.2.2.1.1 tag=2 int32=
column 2 name st
column 2 oid 1.3.6.1.2.1.2.2.1.8 tag=2 int32=
column 2 replace admin 2
column 2 replace down 2
column 2 replace up 1
column 3 name bw
column 3 oid 1.3.6.1.2.1.2.2.1.5 tag=2 class=1 int32=
column 3 script puts [expr $self/1000000]
column 3 oid 1.3.6.1.2.1.31.1.1.1.15 tag=2 class=1 int32=
column 4 name mtu
column 4 oid 1.3.6.1.2.1.2.2.1.4 tag=2 int32=
.
exit
